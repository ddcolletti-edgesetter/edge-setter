The framework you want is Dempster-Shafer Theory (DST).
It's a general framework for reasoning with uncertainty — specifically designed to combine evidence from multiple independent sources into a single degree of belief. It was built for exactly your problem: you have N independent signals of varying quality, and you want to combine them into a single confidence score that gets stronger as evidence accumulates. Wikipedia
Here's why it fits EdgeSetter better than a hardcoded threshold:
Instead of asking "did we get 2 supporting signals from tier1?" it asks "what is the combined belief mass across all evidence, weighted by source reliability?" Each piece of evidence — a tier1 report, a tier3 rumor, a corroborating API result — contributes a belief mass between 0 and 1. They combine multiplicatively, not additively. A single strong source gets you to maybe 0.80. Two independent strong sources get you to 0.96. Three get you to 0.99. The math handles the threshold question for you.
The key property for EdgeSetter: DST separates belief (confidence the hypothesis is true), plausibility (how likely it might still be true), and doubt (how likely it is false). That maps directly onto your verdict states — rumor, likely, confirmed — with confidence being the belief mass. GeeksforGeeks
In practice for your Verifier, it looks like this:
typescriptfunction dempsterCombine(sources: Array<{ beliefMass: number }>): number {
  // Each source contributes a belief mass based on its tier + stance
  // Sources combine multiplicatively — two independent 0.8s = 0.96
  let combined = sources[0].beliefMass;
  for (let i = 1; i < sources.length; i++) {
    combined = 1 - (1 - combined) * (1 - sources[i].beliefMass);
  }
  return combined;
}

// Dempster's Rule of Combination:
//   combined = 1 - (1 - m1)(1 - m2)...(1 - mN)
//
// Belief masses represent how much each source REDUCES remaining doubt.
// They are multiplicative, not additive — two 0.85 sources combine to
// 1 - (0.15 × 0.15) = 0.9775, not 1.70.
//
// Tier belief masses are calibrated starting priors. Once the historical
// bootstrap runs, these should be replaced with real per-source accuracy
// scores from the source scorer.

export type SourceTier = "tier1" | "tier2" | "tier3" | "tier4" | "tier5";

export interface EvidenceSource {
  tier: SourceTier;
  stance: "support" | "contradict" | "neutral";
  sourceAccuracyOverride?: number; // 0-1, from source scorer if available
}

// Base belief masses by tier — starting priors.
// Tune these against historical outcomes after bootstrap runs.
const TIER_BELIEF_MASS: Record<SourceTier, number> = {
  tier1: 0.85,
  tier2: 0.65,
  tier3: 0.40,
  tier4: 0.22,
  tier5: 0.12,
};

// These drive the if/else verdict assignment — the only hardcoded thresholds.
export const CONFIDENCE_THRESHOLDS = {
  VERIFIED:    0.95,  // agent consensus is definitive — show VERIFIED / 100%
  CONFIRMED:   0.85,  // strong consensus, still escalating
  LIKELY:      0.65,  // meaningful signal, multiple sources
  DEVELOPING:  0.45,  // early detection, single or weak sources
  // below DEVELOPING = rumor
};

export interface DSTResult {
  belief: number;         // 0-1 combined belief mass
  confidence: number;     // 0-100 integer for display + storage
  conflictMass: number;   // 0-1 how much evidence is in conflict
  supportCount: number;
  contradictCount: number;
}

  const contradicting = sources.filter(s => s.stance === "contradict");

  const supportCount    = supporting.length;
  const contradictCount = contradicting.length;

  if (supportCount === 0 && contradictCount === 0) {
    return { belief: 0, confidence: 0, conflictMass: 0, supportCount: 0, contradictCount: 0 };
  }

  // Combine supporting evidence using Dempster's rule
  let supportBelief = 0;
  if (supporting.length > 0) {
    // Use sourceAccuracyOverride from source scorer if available,
    // otherwise fall back to tier-based prior
    const firstMass = supporting[0].sourceAccuracyOverride
      ?? TIER_BELIEF_MASS[supporting[0].tier];
    supportBelief = firstMass;

    for (let i = 1; i < supporting.length; i++) {
      const mass = supporting[i].sourceAccuracyOverride
        ?? TIER_BELIEF_MASS[supporting[i].tier];
      supportBelief = 1 - (1 - supportBelief) * (1 - mass);
    }
  }

  // Combine contradicting evidence the same way
  let contradictBelief = 0;
  if (contradicting.length > 0) {
    const firstMass = contradicting[0].sourceAccuracyOverride
      ?? TIER_BELIEF_MASS[contradicting[0].tier];
    contradictBelief = firstMass;

    for (let i = 1; i < contradicting.length; i++) {
      const mass = contradicting[i].sourceAccuracyOverride
        ?? TIER_BELIEF_MASS[contradicting[i].tier];
      contradictBelief = 1 - (1 - contradictBelief) * (1 - mass);
    }
  }

  // Conflict mass: how much support and contradiction directly oppose each other
  const conflictMass = supportBelief * contradictBelief;

  // Net belief after conflict is subtracted and renormalized
  // Classic DST normalization: K = 1 - conflictMass
  const K = 1 - conflictMass;
  const netBelief = K > 0 ? (supportBelief * (1 - contradictBelief)) / K : 0;

  const confidence = Math.round(Math.min(netBelief, 1) * 100);

  return {
    belief: netBelief,
    confidence,
    conflictMass,
    supportCount,
    contradictCount,
  };
}

// Map a DST result to a verdict string + needsReview flag
export function verdictFromDST(result: DSTResult): {
  verdict: "confirmed" | "likely" | "rumor" | "contradicted" | "review";
  needsReview: boolean;
  displayConfidence: number;
} {
  const { belief, conflictMass, confidence, contradictCount } = result;

  // High conflict mass = genuine signal war — route to review
  if (conflictMass > 0.30) {
    return { verdict: "review", needsReview: true, displayConfidence: confidence };
  }

  // Net contradicted — more doubt mass than belief mass
  if (contradictCount > 0 && belief < 0.35) {
    return { verdict: "contradicted", needsReview: false, displayConfidence: confidence };
  }

  if (belief >= CONFIDENCE_THRESHOLDS.VERIFIED) {
    // Agent consensus is definitive. North Star: this IS verification.
    return { verdict: "confirmed", needsReview: false, displayConfidence: 100 };
  }

  if (belief >= CONFIDENCE_THRESHOLDS.CONFIRMED) {
    return { verdict: "confirmed", needsReview: false, displayConfidence: confidence };
  }

  if (belief >= CONFIDENCE_THRESHOLDS.LIKELY) {
    return { verdict: "likely", needsReview: false, displayConfidence: confidence };
  }

  if (belief >= CONFIDENCE_THRESHOLDS.DEVELOPING) {
    return { verdict: "likely", needsReview: false, displayConfidence: confidence };
  }

  return { verdict: "rumor", needsReview: false, displayConfidence: confidence };
}

  }

  // ── Cap and finalize ──────────────────────────────────────────────────────
  // insiderScore is capped at 0.60 — even the strongest local insider
  // shouldn't exceed a well-corroborated tier1 source on their own.
  insiderScore = Math.min(insiderScore, 0.60);

  // Translate insider score into a DST mass boost.
  // A tier3 source has base mass 0.40. With a full insider boost of 0.60,
  // their effective mass reaches 0.88 — just under a lone tier1 (0.85).
  // That's the right ceiling: a local insider alone isn't VERIFIED,
  // but they're close enough to push a second supporting source over the threshold.
  const effectiveMassBoost = insiderScore * 0.50;

  const isInsider = insiderScore >= 0.20;

  return {
    isInsider,
    insiderScore,
    effectiveMassBoost,
    reasons,
  };
}

// Convenience: check if a claim text contains first-reporter language
// without building a full InsiderContext. Used by ingestion layer.
export function hasInsiderLanguage(text: string): boolean {
  return INSIDER_LANGUAGE_PATTERNS.some(p => p.test(text));
}

typescript// Get team-specific accuracy for a source on a given team.
// Returns null if no history exists for this source+team combination.
export async function getSourceTeamAccuracy(
  sourceId: string,
  teamKey: string,
): Promise<{ accuracy: number; sampleSize: number } | null> {
  const row = await db
    .select({ team_accuracies: sourceScores.team_accuracies })
    .from(sourceScores)
    .where(eq(sourceScores.source_id, sourceId))
    .limit(1);

  if (!row.length || !row[0].team_accuracies) return null;

  try {
    const parsed = JSON.parse(row[0].team_accuracies as string);
    const entry = parsed[teamKey];
    if (!entry) return null;
    return { accuracy: entry.accuracy, sampleSize: entry.sample_size };
  } catch {
    return null;
  }
}

// Update team-specific accuracy for a source after an outcome is confirmed.
// Called by sourceScorerOnOutcome when a claim resolves.
export async function updateSourceTeamAccuracy(
  sourceId: string,
  teamKey: string,
  wasCorrect: boolean,
): Promise<void> {
  const row = await db
    .select({ team_accuracies: sourceScores.team_accuracies })
    .from(sourceScores)
    .where(eq(sourceScores.source_id, sourceId))
    .limit(1);

  let parsed: Record<string, { accuracy: number; sample_size: number }> = {};

  if (row.length && row[0].team_accuracies) {
    try { parsed = JSON.parse(row[0].team_accuracies as string); } catch { /* start fresh */ }
  }

  const existing = parsed[teamKey] ?? { accuracy: 0, sample_size: 0 };
  const newSampleSize = existing.sample_size + 1;
  // Running average
  const newAccuracy =
    (existing.accuracy * existing.sample_size + (wasCorrect ? 1 : 0)) / newSampleSize;

  parsed[teamKey] = { accuracy: newAccuracy, sample_size: newSampleSize };

  await db
    .update(sourceScores)
    .set({ team_accuracies: JSON.stringify(parsed) })
    .where(eq(sourceScores.source_id, sourceId));
}

      : undefined;

typescriptexport interface EvidenceSource {
  tier: SourceTier;
  stance: "support" | "contradict" | "neutral";
  sourceAccuracyOverride?: number;
  insiderMassBoost?: number;  // additive boost from insider signal detection

function effectiveMass(source: EvidenceSource): number {
  const base = source.sourceAccuracyOverride ?? TIER_BELIEF_MASS[source.tier];
  const boost = source.insiderMassBoost ?? 0;
  // Cap at 0.88 — a lone insider should never hit VERIFIED on their own.
  // VERIFIED requires either two strong sources or one source + corroboration.
  return Math.min(base + boost, 0.88);
}

export async function countRecentSignalsForPlayerTopic(
  player: string | null,
  eventType: string,
  league: string,
  withinMinutes: number,
): Promise<number> {
  if (!player) return 0;
  const since = new Date(Date.now() - withinMinutes * 60 * 1000).toISOString();

  const result = await db
    .select({ count: count() })
    .from(rawEvents)
    .where(
      and(
        eq(rawEvents.player, player),
        eq(rawEvents.event_type, eventType),
        eq(rawEvents.league, league),
        gte(rawEvents.created_at, since),
      )
    );

  return result[0]?.count ?? 0;
}
